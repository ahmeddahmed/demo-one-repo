#include <stdio.h>

int main(void)
{
    int choice = 2;

    switch (choice) {
        case 1:
            printf("You chose Start\n");
            break;

        case 2:
            printf("You chose Settings\n");
            break;

        case 3:
            printf("You chose Exit\n");
            break;

        default:
            printf("Invalid choice\n");
            break;
    }
    for (int i = 1; i <= 5; i++) {
        printf("%d\n", i);
    }
    return 0;
}
